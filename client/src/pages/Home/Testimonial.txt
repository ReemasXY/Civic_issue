const extendedTestimonials = [
  ...testimonials,
  ...testimonials,
];

const CARD_WIDTH = 380;
const VISIBLE_CARDS = 3;

const TestimonialCarousel = () => {
  const [offset, setOffset] = useState(0);
  const [currentCenter, setCurrentCenter] = useState(0);
  const [isPaused, setIsPaused] = useState(false);

  const frameRef = useRef(null);

  const speed = 0.7;
  const pauseTime = 2500;

  // ==========================================
  // AUTO SLIDE
  // ==========================================

  useEffect(() => {
    if (isPaused) return;

    let pause = false;
    let pauseTimeout;

    const tick = () => {
      if (!pause) {
        setOffset((prev) => {
          let next = prev + speed;

          const maxOffset =
            CARD_WIDTH * testimonials.length;

          if (next >= maxOffset) {
            next = 0;
          }

          const centerIndex =
            Math.floor(
              (next +
                CARD_WIDTH *
                  (VISIBLE_CARDS / 2)) /
                CARD_WIDTH
            ) % testimonials.length;

          setCurrentCenter(centerIndex);

          if (next % CARD_WIDTH < speed) {
            pause = true;

            pauseTimeout = setTimeout(() => {
              pause = false;
            }, pauseTime);
          }

          return next;
        });
      }

      frameRef.current =
        requestAnimationFrame(tick);
    };

    frameRef.current =
      requestAnimationFrame(tick);

    return () => {
      cancelAnimationFrame(frameRef.current);
      clearTimeout(pauseTimeout);
    };
  }, [isPaused]);

  // ==========================================
  // PREVIOUS
  // ==========================================

  const prev = () => {
    setOffset((previous) => {
      let next = previous - CARD_WIDTH;

      const maxOffset =
        CARD_WIDTH * testimonials.length;

      if (next < 0) {
        next =
          maxOffset - CARD_WIDTH;
      }

      const centerIndex =
        Math.floor(
          (next +
            CARD_WIDTH *
              (VISIBLE_CARDS / 2)) /
            CARD_WIDTH
        ) % testimonials.length;

      setCurrentCenter(centerIndex);

      return next;
    });
  };

  // ==========================================
  // NEXT
  // ==========================================

  const next = () => {
    setOffset((previous) => {
      let next = previous + CARD_WIDTH;

      const maxOffset =
        CARD_WIDTH * testimonials.length;

      if (next >= maxOffset) {
        next = 0;
      }

      const centerIndex =
        Math.floor(
          (next +
            CARD_WIDTH *
              (VISIBLE_CARDS / 2)) /
            CARD_WIDTH
        ) % testimonials.length;

      setCurrentCenter(centerIndex);

      return next;
    });
  };

  // ==========================================
  // STAR RATING
  // ==========================================

  const StarRating = ({ rating }) => (
    <div className="mb-4 flex justify-center gap-1">
      {[...Array(5)].map((_, index) => (
        <svg
          key={index}
          className={`h-4 w-4 ${
            index < rating
              ? "text-yellow-400"
              : "text-gray-300"
          }`}
          fill="currentColor"
          viewBox="0 0 20 20"
        >
          <path d="M9.049 2.927c.3-.921 1.603-.921 1.902 0l1.07 3.292a1 1 0 00.95.69h3.462c.969 0 1.371 1.24.588 1.81l-2.8 2.034a1 1 0 00-.364 1.118l1.07 3.292c.3.921-.755 1.688-1.54 1.118l-2.8-2.034a1 1 0 00-1.175 0l-2.8 2.034c-.784.57-1.838-.197-1.539-1.118l1.07-3.292a1 1 0 00-.364-1.118L2.98 8.72c-.783-.57-.38-1.81.588-1.81h3.461a1 1 0 00.951-.69l1.07-3.292z" />
        </svg>
      ))}
    </div>
  );

  return (
    <section className="relative overflow-hidden bg-slate-50 py-20">

      {/* Soft Background Decoration */}
      <div className="pointer-events-none absolute inset-0 overflow-hidden">
        <div className="absolute -right-20 top-20 h-40 w-40 rounded-full bg-teal-100/40 blur-3xl" />

        <div className="absolute -bottom-20 -left-20 h-40 w-40 rounded-full bg-teal-100/40 blur-3xl" />
      </div>

      <div className="relative mx-auto max-w-[1150px] px-5 sm:px-8">

        {/* ==========================================
            SECTION HEADER
        ========================================== */}

        <div className="mb-14 text-center">

          {/* Eyebrow */}
          <div className="mb-5 flex items-center justify-center gap-4">

            <span className="h-[2px] w-10 bg-teal-600" />

            <span className="text-[13px] font-semibold tracking-[2.5px] text-teal-600 sm:text-[14px]">
              CITIZEN VOICES
            </span>

            <span className="h-[2px] w-10 bg-teal-600" />

          </div>

          {/* Heading */}
          <h2 className="text-[42px] font-bold leading-[1.1] tracking-[-1.5px] text-[#14233B] sm:text-[48px] lg:text-[52px]">

            What Citizens
            <br />

            <span className="text-teal-600">
              Are Saying
            </span>

          </h2>

          {/* Description */}
          <p className="mx-auto mt-6 max-w-[650px] text-[16px] leading-[1.7] text-slate-600 sm:text-[17px]">
            Real experiences from citizens using our platform
            to report, track, and help resolve civic issues in
            their communities.
          </p>

        </div>

        {/* ==========================================
            CAROUSEL
        ========================================== */}

        <div
          className="relative mx-auto max-w-7xl"
          onMouseEnter={() =>
            setIsPaused(true)
          }
          onMouseLeave={() =>
            setIsPaused(false)
          }
        >

          <div className="overflow-hidden py-8">

            <div
              className="flex transition-transform duration-500 ease-out"
              style={{
                transform: `translateX(-${offset}px)`,
              }}
            >

              {extendedTestimonials.map(
                (testimonial, index) => {

                  const relativeIndex =
                    index %
                    testimonials.length;

                  const distance = Math.abs(
                    relativeIndex -
                      currentCenter
                  );

                  const scale =
                    distance === 0
                      ? 1.03
                      : distance === 1
                      ? 0.96
                      : 0.90;

                  const opacity =
                    distance <= 1
                      ? 1
                      : 0.45;

                  const zIndex =
                    distance === 0
                      ? 10
                      : 1;

                  return (
                    <div
                      key={index}
                      className="flex-shrink-0 px-4"
                      style={{
                        width: `${CARD_WIDTH}px`,
                        transform: `scale(${scale})`,
                        opacity,
                        zIndex,
                        transition:
                          "all 0.5s cubic-bezier(0.4, 0, 0.2, 1)",
                      }}
                    >

                      {/* ==========================================
                          TESTIMONIAL CARD
                      ========================================== */}

                      <div
                        className={`
                          relative
                          rounded-2xl
                          border
                          border-gray-100
                          bg-white
                          p-8
                          shadow-sm
                          transition-all
                          duration-500
                          ${
                            distance === 0
                              ? "shadow-md"
                              : ""
                          }
                        `}
                      >

                        {/* Quote Icon */}
                        <div className="absolute right-6 top-6 text-teal-100">

                          <svg
                            width="32"
                            height="32"
                            viewBox="0 0 24 24"
                            fill="currentColor"
                          >
                            <path d="M14.017 21v-7.391c0-5.704 3.731-9.57 8.983-10.609l.995 2.151c-2.432.917-3.995 3.638-3.995 5.849h4v10h-9.983zm-14.017 0v-7.391c0-5.704 3.748-9.57 9-10.609l.996 2.151c-2.433.917-3.996 3.638-3.996 5.849h3.983v10h-9.983z" />
                          </svg>

                        </div>

                        {/* ==========================================
                            PROFILE IMAGE
                        ========================================== */}

                        <div className="relative mb-6 flex justify-center">

                          <div className="relative">

                            <div className="absolute inset-0 rounded-full bg-teal-500 p-1">

                              <div className="h-20 w-20 rounded-full bg-white" />

                            </div>

                            <img
                              src={testimonial.image}
                              alt={testimonial.name}
                              className="relative h-20 w-20 rounded-full border-4 border-white object-cover"
                            />

                          </div>

                        </div>

                        {/* Stars */}
                        <StarRating
                          rating={testimonial.rating}
                        />

                        {/* ==========================================
                            MESSAGE
                        ========================================== */}

                        <blockquote className="mb-6 flex min-h-[100px] items-center text-center leading-relaxed">

                          <p className="text-[16px] italic text-slate-600">
                            "{testimonial.message}"
                          </p>

                        </blockquote>

                        {/* ==========================================
                            USER INFO
                        ========================================== */}

                        <div className="text-center">

                          <h3 className="mb-1 text-lg font-bold text-[#14233B]">
                            {testimonial.name}
                          </h3>

                          <p className="text-sm font-medium text-teal-600">
                            {testimonial.role}
                          </p>

                        </div>

                        {/* Bottom Decoration */}
                        <div className="absolute bottom-0 left-0 h-20 w-20 rounded-bl-2xl rounded-tr-2xl bg-teal-50/70" />

                      </div>

                    </div>
                  );
                }
              )}

            </div>

          </div>

          {/* ==========================================
              PREVIOUS BUTTON
          ========================================== */}

          <button
            onClick={prev}
            className="
              absolute
              left-0
              top-1/2
              z-20
              flex
              h-11
              w-11
              -translate-y-1/2
              items-center
              justify-center
              rounded-full
              bg-teal-600
              text-white
              shadow-md
              transition-all
              duration-300
              hover:bg-teal-700
              hover:shadow-lg
              sm:left-2
            "
            aria-label="Previous testimonial"
          >
            <svg
              className="h-5 w-5"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M15 19l-7-7 7-7"
              />
            </svg>
          </button>

          {/* ==========================================
              NEXT BUTTON
          ========================================== */}

          <button
            onClick={next}
            className="
              absolute
              right-0
              top-1/2
              z-20
              flex
              h-11
              w-11
              -translate-y-1/2
              items-center
              justify-center
              rounded-full
              bg-teal-600
              text-white
              shadow-md
              transition-all
              duration-300
              hover:bg-teal-700
              hover:shadow-lg
              sm:right-2
            "
            aria-label="Next testimonial"
          >
            <svg
              className="h-5 w-5"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M9 5l7 7-7 7"
              />
            </svg>
          </button>

        </div>

        {/* ==========================================
            DOTS
        ========================================== */}

        <div className="mt-8 flex justify-center gap-2">

          {testimonials.map((_, index) => (
            <button
              key={index}
              onClick={() => {
                setOffset(
                  index * CARD_WIDTH
                );
                setCurrentCenter(index);
              }}
              className={`
                h-2.5
                rounded-full
                transition-all
                duration-300
                ${
                  index === currentCenter
                    ? "w-7 bg-teal-600"
                    : "w-2.5 bg-slate-300 hover:bg-slate-400"
                }
              `}
              aria-label={`Go to testimonial ${
                index + 1
              }`}
            />
          ))}

        </div>

      </div>
    </section>
  );
};

export default TestimonialCarousel;
